cmake_minimum_required(VERSION 3.16)
project(A_Casa LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(BUILD_GAME "Compilar a janela OpenGL" ON)
add_library(game_logic src/game.c src/world.c src/zombies.c src/grapple.c)
target_include_directories(game_logic PUBLIC src)
if(NOT MSVC)
    target_compile_options(game_logic PRIVATE -Wall -Wextra -Wpedantic)
    target_link_libraries(game_logic PUBLIC m)
endif()
if(BUILD_GAME)
    find_package(OpenGL REQUIRED)
    find_package(GLUT REQUIRED)
    add_executable(a_casa src/main.c src/zombie_render.c src/grapple_render.c)
    target_link_libraries(a_casa PRIVATE game_logic OpenGL::GL OpenGL::GLU GLUT::GLUT)
    add_executable(test_integration tests/test_integration.c src/zombie_render.c src/grapple_render.c)
    target_link_libraries(test_integration PRIVATE game_logic OpenGL::GL OpenGL::GLU GLUT::GLUT)
    if(MSVC)
        target_compile_options(test_integration PRIVATE /UNDEBUG)
    else()
        target_compile_options(test_integration PRIVATE -UNDEBUG)
    endif()
    if(NOT MSVC)
        target_compile_options(a_casa PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endif()
enable_testing()
add_executable(test_game tests/test_game.c)
target_link_libraries(test_game PRIVATE game_logic)
# Asserts precisam continuar ativos mesmo em builds Release.
if(MSVC)
    target_compile_options(test_game PRIVATE /UNDEBUG)
else()
    target_compile_options(test_game PRIVATE -UNDEBUG)
endif()
add_test(NAME game_logic COMMAND test_game)

add_executable(test_zombies tests/test_zombies.c)
target_link_libraries(test_zombies PRIVATE game_logic)
if(MSVC)
    target_compile_options(test_zombies PRIVATE /UNDEBUG)
else()
    target_compile_options(test_zombies PRIVATE -UNDEBUG)
endif()
add_test(NAME zombies COMMAND test_zombies)

add_executable(test_grapple tests/test_grapple.c)
target_link_libraries(test_grapple PRIVATE game_logic)
if(MSVC)
    target_compile_options(test_grapple PRIVATE /UNDEBUG)
else()
    target_compile_options(test_grapple PRIVATE -UNDEBUG)
endif()
add_test(NAME grapple COMMAND test_grapple)
